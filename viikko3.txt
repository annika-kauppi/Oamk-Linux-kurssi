#!/usr/bin/env bash

# Destination paths
TARGET_DIR="/var/www/html/viikko3"
TARGET_FILE="$TARGET_DIR/index.html"
TEMP_FILE=$(mktemp)

# Replace with your actual GitHub repository URL
GITHUB_URL="https://github.com"

# Function to evaluate disk space status
check_disk_status() {
    local usage_percent=$1
    if [ "$usage_percent" -ge 80 ]; then
        echo "WARNING"
    else
        echo "OK"
    fi
}

# Collect server metrics using command substitution and pipes
GENERATED_AT=$(date "+%Y-%m-%d %H:%M:%S %Z")
UPTIME_TEXT=$(uptime -p)
DISK_USAGE=$(df -P / | awk 'NR == 2 {print $5}')
DISK_NUMBER=$(echo "$DISK_USAGE" | tr -d '%')
DISK_STATUS=$(check_disk_status "$DISK_NUMBER")
MEM_TOTAL=$(free -m | awk 'NR == 2 {print $2}')
MEM_AVAILABLE=$(free -m | awk 'NR == 2 {print $7}')

# Generate the HTML report into the temporary file
cat > "$TEMP_FILE" <<EOF

    <section>
        <h2>Commands used</h2>
        <ul>
            <li>
                <code>date</code>:
                displays the current date, time and time zone
                using the format specified in the script.
            </li>
            <li>
                <code>uptime -p</code>:
                displays how long the server has been running.
            </li>
            <li>
                <code>df -P / | awk 'NR == 2 {print \$5}'</code>:
                passes filesystem information to awk, which extracts
                the usage percentage from the second row.
            </li>
            <li>
                <code>free -m</code>:
                displays memory information in mebibytes (MiB).
            </li>
        </ul>

    <section>
        <h2>How the script works</h2>
        <p>
            The script collects server information using Linux commands
            and stores the results in Bash variables through command
            substitution. A function uses an if statement to determine
            the disk space status.
        </p>
        <p>
            The script generates a temporary HTML file and then moves
            the completed file into place as index.html.
            Nginx serves this file at /viikko3/.
        </p>
        <p>
            The <code>check_disk_status</code> function receives the numeric disk usage percentage as an argument and checks whether it is 80 or greater using <code>[ "\$usage_percent" -ge 80 ]</code>. For piping, <code>free -m | awk 'NR == 2 {print \$7}'</code> sends the memory table to <code>awk</code> to extract only the available RAM column.
        </p>
    </section>

    <section>
        <h2>Scheduling</h2>
        <p>
            I used <strong>cron</strong> to schedule the script to run once every minute using the entry <code>* * * * * /home/ubuntu/viikko3/report.sh</code>. Because the cron daemon is enabled as a system service by default on Ubuntu, scheduled jobs resume automatically after a virtual machine reboot.
        </p>
    </section>

        <h2>A problem solved or a lesson learned</h2>
        <p>
            Initially, Nginx returned a <strong>403 Forbidden</strong> error because <code>mktemp</code> creates temporary files with <code>600</code> permissions (readable only by the file owner). Adding <code>chmod 644 "\$TEMP_FILE"</code> before moving the file into <code>/var/www/html/viikko3/index.html</code> allowed Nginx to read and serve the page properly.
